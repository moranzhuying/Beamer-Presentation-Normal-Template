\begin{frame}{两栏布局}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \begin{theorem}{费马大定理}{}
        对整数 $n \geq 3$, 方程
        \[
          x^n + y^n = z^n
        \]
        无非平凡整数解。
      \end{theorem}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{itemize}
        \item 由 Andrew Wiles 于 1995 年证明
        \item 核心工具: 模形式与椭圆曲线
        \item 与谷山--志村猜想等价
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
