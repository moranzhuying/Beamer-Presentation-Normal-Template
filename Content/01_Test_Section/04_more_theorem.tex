\begin{frame}{更多定理环境}
  \begin{lemma}{}
    若 $f$ 在闭区间上连续, 则 $f$ 一致连续。
  \end{lemma}

  \begin{proposition}{}
    连续函数把紧集映射为紧集。
  \end{proposition}

  \begin{remark}{}
    上述结论在度量空间框架下仍然成立。
  \end{remark}
\end{frame}
